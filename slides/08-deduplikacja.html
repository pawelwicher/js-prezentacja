Benchmark — deduplikacja

filter + indexOf  vs  Set

✗  Przed optymalizacją — Array 
// Deduplikacja przez filter + indexOf
// Złożoność: O(n²)
function deduplicateArray(arr) {
  return arr.filter(
    (item, index) => arr.indexOf(item) === index
  );
}

// Tablica 50 000 elementów (połowa duplikatów)
deduplicateArray(data);
✓  Po optymalizacji — Set 
// Deduplikacja przez Set
// Złożoność: O(n)
function deduplicateSet(arr) {
  return [...new Set(arr)];
}

// Tablica 50 000 elementów (połowa duplikatów)
deduplicateSet(data);
